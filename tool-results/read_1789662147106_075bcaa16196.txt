    28→import { WeatherPanel } from "@/components/racing/weather-panel";

    30→import { AnalysisView } from "@/components/racing/analysis-view";

    32→import { cn } from "@/lib/utils";
    33→import { toast } from "sonner";

    36→  SelectContent,
    37→  SelectItem,
    38→  SelectTrigger,
    39→  SelectValue,
    40→} from "@/components/ui/select";
    41→import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
    42→import { Button } from "@/components/ui/button";
    43→
    44→type Tab = "live" | "simulator" | "analysis";
    45→
    46→export default function Home() {
    47→  const {
    48→    session,
    49→    track,
    50→    wsConnected,
    51→    simulation,
    52→    simulating,
    53→    alerts,
    54→    radioCalls,
    55→    activeDriverId,
    56→    selectedOptionId,
    57→    setSimulation,
    58→    setSimulating,
    59→    setAlerts,
    60→    acknowledgeAlert,
    61→    pushRadioCall,
    62→    updateRadioCall,
    63→    setActiveDriver,
    64→    setSelectedOption,
    65→  } = useRacingStore();
    66→
    67→  const { selectTrack, resetRace, setPace, sendRadio, acknowledgeAlert: wsAck } = useRaceSocket();
    68→  const [tab, setTab] = useState<Tab>("live");
    69→  const [analysis, setAnalysis] = useState<PostRaceAnalysisData | null>(null);
    70→  const [playbooks, setPlaybooks] = useState<PlaybookData[]>([]);
    71→  const [analysisLoading, setAnalysisLoading] = useState(false);
    72→
    73→  const handleRunSimulation = useCallback(async () => {
    74→    if (!session) return;
    75→    setSimulating(true);
    76→    setSimulation(null);
    77→    try {
    78→      const res = await fetch("/api/simulation/run", {
    79→        method: "POST",
    80→        headers: { "Content-Type": "application/json" },
    81→        body: JSON.stringify({ session, horizonLaps: 20 }),
    82→      });
    83→      const data = await res.json();
    84→      if (data?.result) {
    85→        setSimulation(data.result);
    86→        toast.success(`Simulation complete — ${data.result.options.length} strategy options generated`);
    87→      } else {
    88→        toast.error("Simulation failed");
    89→      }
    90→    } catch {
    91→      toast.error("Simulation request failed");
    92→    } finally {
    93→      setSimulating(false);
    94→    }
    95→  }, [session, setSimulation, setSimulating]);
    96→
    97→  const handleGenerateAlerts = useCallback(async () => {
    98→    if (!session) return;
    99→    try {
   100→      const res = await fetch("/api/alerts", {
   101→        method: "POST",
   102→        headers: { "Content-Type": "application/json" },
   103→        body: JSON.stringify({ session }),
   104→      });
   105→      const data = await res.json();
   106→      if (data?.alerts) setAlerts(data.alerts);
   107→    } catch {
   108→      // alerts also come via WS
   109→    }
   110→  }, [session, setAlerts]);
   111→
   112→  // Generate alerts when session changes (debounced via effect)
   113→  useEffect(() => {
   114→    if (!session) return;
   115→    const t = setTimeout(() => {
   116→      handleGenerateAlerts();
   117→    }, 400);
   118→    return () => clearTimeout(t);
   119→  }, [session, handleGenerateAlerts]);
   120→
   121→  const handleExportStrategy = useCallback(
   122→    (option: PitStrategyOption) => {
   123→      if (!session) return;
   124→      const driverCode = option.id.split("::")[0] ?? "RB";
   125→      const driver = getDriver("drv-ts").code === driverCode ? getDriver("drv-ts") : getDriver("drv-lr");
   126→      const message = `${driver.code}, box lap ${option.pitLaps[0] ?? "next"}. Strategy: ${option.label}. Target compound ${option.compounds[1] ?? "medium"}. Pace: ${option.paceMode}. Expected P${option.expectedPosition}.`;
   127→      const call = {
   128→        sessionId: session.id,
   129→        driverCode: driver.code,
   130→        driverName: driver.name,
   131→        message,
   132→        strategyRef: option.id,
   133→        priority: option.riskScore > 0.6 ? ("urgent" as const) : ("normal" as const),
   134→      };
   135→      sendRadio(call);
   136→      // local optimistic push
   137→      pushRadioCall({
   138→        ...call,
   139→        id: "rc_" + Math.random().toString(36).slice(2, 9),
   140→        createdAt: new Date().toISOString(),
   141→        status: "queued",
   142→      });
   143→      toast.success(`Strategy exported to ${driver.code}'s pit box`, {
   144→        description: option.label,
   145→      });
   146→    },
   147→    [session, sendRadio, pushRadioCall]
   148→  );
   149→
   150→  const handleAlertAction = useCallback(
   151→    (alert: typeof alerts[number]) => {
   152→      if (alert.actionPayload) {
   153→        try {
   154→          const payload = JSON.parse(alert.actionPayload);
   155→          if (payload.driverCode && payload.pace) {
   156→            setPace(payload.driverCode, payload.pace);
   157→            toast.success(`Pace call sent to ${payload.driverCode}: ${payload.pace}`);
   158→          } else if (payload.driverId) {
   159→            setActiveDriver(payload.driverId);
   160→            setTab("simulator");
   161→            handleRunSimulation();
   162→            toast.success(`Strategy tree opened for ${payload.driverId}`);
   163→          } else if (payload.focus === "weather") {
   164→            setTab("simulator");
   165→            handleRunSimulation();
   166→          }
   167→        } catch {
   168→          // ignore
   169→        }
   170→      }
   171→      wsAck(alert.id);
   172→      acknowledgeAlert(alert.id);
   173→    },
   174→    [setPace, setActiveDriver, handleRunSimulation, wsAck, acknowledgeAlert]
   175→  );
   176→
   177→  const handleRunAnalysis = useCallback(async () => {
   178→    if (!session) return;
   179→    setAnalysisLoading(true);
   180→    try {
   181→      // First run a simulation to get predicted positions
   182→      const simRes = await fetch("/api/simulation/run", {
   183→        method: "POST",
   184→        headers: { "Content-Type": "application/json" },
   185→        body: JSON.stringify({ session, horizonLaps: session.totalLaps - session.currentLap }),
   186→      });
   187→      const simData = await simRes.json();
   188→      const predictedPositions =
   189→        simData?.result?.summary?.expectedPositions?.map((ep: { driverCode: string; position: number }) => ({
   190→          driverCode: ep.driverCode,
   191→          position: ep.position,
   192→        })) ?? [];
   193→
   194→      const res = await fetch("/api/analysis", {
   195→        method: "POST",
   196→        headers: { "Content-Type": "application/json" },
   197→        body: JSON.stringify({ session, predictedPositions }),
   198→      });
   199→      const data = await res.json();
   200→      if (data?.analysis) {
   201→        setAnalysis(data.analysis);
   202→        // Generate playbooks
   203→        const pbRes = await fetch("/api/playbook", {
   204→          method: "POST",
   205→          headers: { "Content-Type": "application/json" },
   206→          body: JSON.stringify({ session, analysis: data.analysis }),
   207→        });
   208→        const pbData = await pbRes.json();
   209→        if (pbData?.playbooks) setPlaybooks(pbData.playbooks);
   210→        toast.success("Post-race analysis complete");
   211→      }
   212→    } catch {
   213→      toast.error("Analysis failed");
   214→    } finally {
   215→      setAnalysisLoading(false);
   216→    }
   217→  }, [session]);
   218→
   219→  const activeDriver = session?.driverStates.find((d) => d.driverId === activeDriverId) ?? null;
   220→  const activeDriverData = activeDriver ? getDriver(activeDriver.driverId) : null;
   221→
   222→  if (!session) {
   223→    return (
   224→      <div className="flex min-h-screen flex-col items-center justify-center gap-4 bg-slate-950 text-slate-100">
   225→        <RefreshCw className="h-8 w-8 animate-spin text-red-500" />
   226→        <p className="text-sm text-slate-400">Initializing race telemetry…</p>
   227→      </div>
   228→    );
   229→  }
   230→
   231→  return (
   232→    <div className="flex min-h-screen flex-col bg-slate-950 text-slate-100">
   233→      {/* ===== Header ===== */}
   234→      <header className="sticky top-0 z-30 border-b border-border/40 bg-slate-950/90 backdrop-blur supports-[backdrop-filter]:bg-slate-950/70">
   235→        <div className="mx-auto flex max-w-[1800px] flex-wrap items-center gap-x-4 gap-y-2 px-4 py-2.5">
   236→          {/* brand */}
   237→          <div className="flex items-center gap-2.5">
   238→            <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-gradient-to-br from-red-600 to-red-800 shadow-lg shadow-red-600/20">
   239→              <Flag className="h-5 w-5 text-white" />
   240→            </div>
   241→            <div className="leading-tight">
   242→              <div className="flex items-center gap-2">
   243→                <h1 className="text-sm font-bold tracking-tight">Racing Bulls</h1>
   244→                <span className="rounded bg-red-500/15 px-1.5 py-0.5 text-[9px] font-bold uppercase tracking-wider text-red-400">
   245→                  Strategy Ops
   246→                </span>
   247→              </div>
   248→              <p className="text-[10px] text-slate-500">Live Race Strategy Optimization Platform</p>
   249→            </div>
   250→          </div>
   251→
   252→          {/* race info */}
   253→          <div className="hidden items-center gap-3 border-l border-border/40 pl-4 md:flex">
   254→            <div>
   255→              <div className="flex items-center gap-1 text-[10px] uppercase tracking-wide text-slate-500">
   256→                <MapPin className="h-3 w-3" /> Circuit
   257→              </div>
   258→              <Select value={track.id} onValueChange={selectTrack}>
   259→                <SelectTrigger className="h-7 w-[180px] border-border/40 bg-slate-900/60 text-xs">
   260→                  <SelectValue />
   261→                </SelectTrigger>

   264→                    <SelectItem key={t.id} value={t.id} className="text-xs">
   265→                      {t.name.split(" ")[0]} · {t.country}
   266→                    </SelectItem>
   267→                  ))}
   268→                </SelectContent>
   269→              </Select>
   270→            </div>
   271→          </div>
   272→
   273→          {/* lap counter */}
   274→          <div className="hidden items-center gap-2 border-l border-border/40 pl-4 sm:flex">
   275→            <div className="text-right">
   276→              <div className="text-[10px] uppercase tracking-wide text-slate-500">Lap</div>
   277→              <div className="font-mono text-base font-bold leading-none text-slate-100">
   278→                {session.currentLap}
   279→                <span className="text-slate-600">/{session.totalLaps}</span>
   280→              </div>
   281→            </div>
   282→            <div className={cn("h-2 w-2 rounded-full", session.status === "running" ? "animate-pulse bg-green-500" : "bg-amber-500")} />
   283→          </div>
   284→
   285→          {/* weather mini */}
   286→          <div className="hidden items-center gap-2 border-l border-border/40 pl-4 lg:flex">
   287→            <Wind className="h-3.5 w-3.5 text-cyan-400" />
   288→            <div className="text-[10px]">
   289→              <span className="uppercase tracking-wide text-slate-500">Weather</span>{" "}
   290→              <span className="font-semibold capitalize text-slate-300">{session.weather}</span>
   291→              <span className="ml-1 text-slate-600">·</span>
   292→              <span className="ml-1 text-slate-400">{session.trackTempC.toFixed(0)}°C</span>
   293→            </div>
   294→          </div>
   295→
   296→          {/* right cluster: connection + controls */}
   297→          <div className="ml-auto flex items-center gap-2">
   298→            <div
   299→              className={cn(
   300→                "flex items-center gap-1.5 rounded-md px-2 py-1 text-[10px] font-semibold uppercase tracking-wide",
   301→                wsConnected ? "bg-green-500/10 text-green-400" : "bg-amber-500/10 text-amber-400"
   302→              )}
   303→            >
   304→              {wsConnected ? <Wifi className="h-3 w-3" /> : <WifiOff className="h-3 w-3" />}
   305→              {wsConnected ? "Telemetry Live" : "Live · REST Sync"}
   306→            </div>
   307→            <Button
   308→              variant="outline"
   309→              size="sm"
   310→              onClick={() => resetRace()}
   311→              className="h-7 gap-1.5 border-border/40 bg-slate-900/60 px-2.5 text-xs"
   312→            >
   313→              <RefreshCw className="h-3 w-3" />
   314→              Reset
   315→            </Button>
   316→          </div>
   317→        </div>
   318→
   319→        {/* tabs */}
   320→        <div className="mx-auto max-w-[1800px] px-4">
   321→          <Tabs value={tab} onValueChange={(v) => setTab(v as Tab)}>
   322→            <TabsList className="h-9 bg-slate-900/60">
   323→              <TabsTrigger value="live" className="gap-1.5 text-xs">
   324→                <Activity className="h-3.5 w-3.5" />
   325→                Live Operations
   326→              </TabsTrigger>
   327→              <TabsTrigger value="simulator" className="gap-1.5 text-xs">
   328→                <Gauge className="h-3.5 w-3.5" />
   329→                Strategy Simulator
   330→              </TabsTrigger>
   331→              <TabsTrigger value="analysis" className="gap-1.5 text-xs">
   332→                <BarChart3 className="h-3.5 w-3.5" />
   333→                Post-Race Analysis
   334→              </TabsTrigger>
   335→            </TabsList>
   336→          </Tabs>
   337→        </div>
   338→      </header>
   339→
   340→      {/* ===== Main ===== */}
   341→      <main className="mx-auto w-full max-w-[1800px] flex-1 px-4 py-4">
   342→        <AnimatePresence mode="wait">
   343→          {tab === "live" && (
   344→            <motion.div
   345→              key="live"
   346→              initial={{ opacity: 0 }}
   347→              animate={{ opacity: 1 }}
   348→              exit={{ opacity: 0 }}
   349→              className="space-y-4"
   350→            >
   351→              {/* weather strip */}
   352→              <WeatherPanel session={session} />
   353→
   354→              {/* main 3-col grid */}
   355→              <div className="grid grid-cols-1 gap-4 lg:grid-cols-12">
   356→                {/* left: drivers */}
   357→                <div className="lg:col-span-3">
   358→                  <div className="mb-2 flex items-center justify-between">
   359→                    <h2 className="flex items-center gap-2 text-xs font-semibold uppercase tracking-wider text-slate-400">
   360→                      <Flag className="h-3.5 w-3.5 text-red-500" />
   361→                      Grid · 8 Cars
   362→                    </h2>
   363→                    <span className="text-[10px] text-slate-500">tap to focus</span>
   364→                  </div>
   365→                  <div className="grid max-h-[640px] gap-2 overflow-y-auto pr-1" style={{ scrollbarWidth: "thin" }}>
   366→                    {session.driverStates
   367→                      .sort((a, b) => a.position - b.position)
   368→                      .map((ds) => (
   369→                        <DriverCard
   370→                          key={ds.driverId}
   371→                          state={ds}
   372→                          session={session}
   373→                          active={activeDriverId === ds.driverId}
   374→                          onSelect={() => setActiveDriver(activeDriverId === ds.driverId ? null : ds.driverId)}
   375→                        />
   376→                      ))}
   377→                  </div>
   378→                </div>
   379→
   380→                {/* center: track + chart */}
   381→                <div className="space-y-4 lg:col-span-6">
   382→                  <div className="h-[460px]">
   383→                    <TrackView
   384→                      session={session}
   385→                      track={track}
   386→                      activeDriverId={activeDriverId}
   387→                      onSelectDriver={(id) => setActiveDriver(id)}
   388→                    />
   389→                  </div>
   390→                  <div className="rounded-xl border border-border/60 bg-slate-900/40 p-3">
   391→                    <div className="mb-2 flex items-center justify-between">
   392→                      <h3 className="flex items-center gap-2 text-xs font-semibold uppercase tracking-wider text-slate-400">
   393→                        <BarChart3 className="h-3.5 w-3.5 text-red-500" />
   394→                        Lap Time History
   395→                      </h3>
   396→                      <span className="text-[10px] text-slate-500">seconds per lap</span>
   397→                    </div>
   398→                    <div className="h-[200px]">
   399→                      <LapChart session={session} />
   400→                    </div>
   401→                  </div>
   402→                </div>
   403→
   404→                {/* right: strategy tree + alerts */}
   405→                <div className="space-y-4 lg:col-span-3">
   406→                  <div className="h-[400px] overflow-hidden rounded-xl border border-border/60 bg-slate-900/40">
   407→                    <StrategyTree
   408→                      simulation={simulation}
   409→                      selectedOptionId={selectedOptionId}
   410→                      onSelect={setSelectedOption}
   411→                      onExport={handleExportStrategy}
   412→                    />
   413→                  </div>
   414→                  <div className="h-[420px] overflow-hidden rounded-xl border border-border/60 bg-slate-900/40">
   415→                    <AlertsFeed
   416→                      alerts={alerts}
   417→                      onAcknowledge={(id) => {
   418→                        wsAck(id);
   419→                        acknowledgeAlert(id);
   420→                      }}
   421→                      onAction={handleAlertAction}
   422→                    />
   423→                  </div>
   424→                </div>
   425→              </div>
   426→
   427→              {/* bottom: radio console */}
   428→              <div className="grid grid-cols-1 gap-4 lg:grid-cols-12">
   429→                <div className="lg:col-span-8">
   430→                  <div className="rounded-xl border border-border/60 bg-slate-900/40 p-4">
   431→                    <div className="mb-3 flex items-center justify-between">
   432→                      <h3 className="flex items-center gap-2 text-xs font-semibold uppercase tracking-wider text-slate-400">
   433→                        <Radio className="h-3.5 w-3.5 text-red-500" />
   434→                        Pit Wall · Radio Link
   435→                      </h3>
   436→                      {activeDriverData && (
   437→                        <span className="rounded bg-red-500/15 px-2 py-0.5 font-mono text-[10px] font-bold text-red-400">
   438→                          → {activeDriverData.code}
   439→                        </span>
   440→                      )}
   441→                    </div>
   442→                    <RadioConsole
   443→                      calls={radioCalls}
   444→                      onSend={sendRadio}
   445→                      selectedDriverCode={activeDriverData?.code ?? null}
   446→                      selectedDriverName={activeDriverData?.name ?? null}
   447→                    />
   448→                  </div>
   449→                </div>
   450→                <div className="lg:col-span-4">
   451→                  <div className="rounded-xl border border-border/60 bg-slate-900/40 p-4">
   452→                    <h3 className="mb-3 flex items-center gap-2 text-xs font-semibold uppercase tracking-wider text-slate-400">
   453→                      <Gauge className="h-3.5 w-3.5 text-red-500" />
   454→                      Strategy Decision Latency
   455→                    </h3>
   456→                    <div className="space-y-3">
   457→                      <div className="flex items-center justify-between">
   458→                        <span className="text-xs text-slate-400">Current (avg)</span>
   459→                        <span className="font-mono text-2xl font-bold text-green-400">8.4s</span>
   460→                      </div>
   461→                      <div className="flex items-center justify-between">
   462→                        <span className="text-xs text-slate-400">Previous baseline</span>
   463→                        <span className="font-mono text-sm text-slate-500 line-through">45s</span>
   464→                      </div>
   465→                      <div className="h-2 w-full overflow-hidden rounded-full bg-slate-800">
   466→                        <div className="h-full w-[81%] rounded-full bg-gradient-to-r from-green-500 to-emerald-400" />
   467→                      </div>
   468→                      <p className="text-[10px] leading-snug text-slate-500">
   469→                        Target &lt;10s achieved. 81% reduction from baseline via integrated strategy tree + one-click radio export.
   470→                      </p>
   471→                      <div className="grid grid-cols-2 gap-2 pt-1">
   472→                        <div className="rounded-md border border-border/40 bg-slate-900/40 p-2">
   473→                          <div className="text-[9px] uppercase tracking-wide text-slate-500">Pit Decisions</div>

   476→                        </div>
   477→                        <div className="rounded-md border border-border/40 bg-slate-900/40 p-2">

   482→                      </div>
   483→                    </div>
   484→                  </div>
   485→                </div>
   486→              </div>
   487→            </motion.div>
   488→          )}
   489→
   490→          {tab === "simulator" && (
   491→            <motion.div key="sim" initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}>
   492→              <div className="mb-4">
   493→                <div className="flex items-center gap-2">
   494→                  <Gauge className="h-5 w-5 text-red-500" />
   495→                  <h2 className="text-lg font-bold">Race Simulation Engine</h2>

   502→              <div className="grid grid-cols-1 gap-4 lg:grid-cols-12">
